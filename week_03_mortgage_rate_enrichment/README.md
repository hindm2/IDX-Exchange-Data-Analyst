from pathlib import Path
import zipfile

# Updated to match the exact file name with the space
zip_path = Path("/content/IDX_Analysis.zip")
extract_path = Path("/content/IDX_Analysis")

with zipfile.ZipFile(zip_path, "r") as zip_ref:
    zip_ref.extractall(extract_path)

csv_files = [
    p for p in extract_path.rglob("*.csv")
    if "__MACOSX" not in p.parts and not p.name.startswith("._")
]

print(f"Extracted {len(csv_files)} CSV files.")

from pathlib import Path
import pandas as pd

folder = Path("/content/IDX_Analysis")

sold_files = sorted(folder.rglob("CRMLSSold*.csv"))
listing_files = sorted(folder.rglob("CRMLSListing*.csv"))

# Ignore Mac metadata files
sold_files = [f for f in sold_files if "__MACOSX" not in f.parts]
listing_files = [f for f in listing_files if "__MACOSX" not in f.parts]

print("Sold files:", len(sold_files))
print("Listing files:", len(listing_files))

assert len(sold_files) == 33, "Expected 33 sold files."
assert len(listing_files) == 33, "Expected 33 listing files."

sold = pd.concat(
    [pd.read_csv(f, low_memory=False) for f in sold_files],
    ignore_index=True
)

listings = pd.concat(
    [pd.read_csv(f, low_memory=False) for f in listing_files],
    ignore_index=True
)

print("\nCombined sold rows:", f"{len(sold):,}")
print("Combined listing rows:", f"{len(listings):,}")

# Check the date columns needed for the mortgage-rate merge
assert "CloseDate" in sold.columns, "CloseDate is missing."
assert "ListingContractDate" in listings.columns, "ListingContractDate is missing."

url = "https://fred.stlouisfed.org/graph/fredgraph.csv?id=MORTGAGE30US"

mortgage = pd.read_csv(url)
mortgage.columns = ["date", "rate_30yr_fixed"]

mortgage["date"] = pd.to_datetime(mortgage["date"])
mortgage["rate_30yr_fixed"] = pd.to_numeric(
    mortgage["rate_30yr_fixed"], errors="coerce"
)

mortgage["year_month"] = mortgage["date"].dt.to_period("M")

mortgage_monthly = (
    mortgage.groupby("year_month", as_index=False)["rate_30yr_fixed"]
    .mean()
)

# Preview the months covered by your MLS files
print(
    mortgage_monthly[
        mortgage_monthly["year_month"].between("2024-01", "2026-09")
    ].to_string(index=False)
)

"""
The monthly average 30-year fixed mortgage rate ranged from 6.05% to 7.06% between January 2024 and September 2026, ending at 6.86%."""

# Create monthly keys from the appropriate transaction dates
sold["year_month"] = pd.to_datetime(
    sold["CloseDate"], errors="coerce"
).dt.to_period("M")

listings["year_month"] = pd.to_datetime(
    listings["ListingContractDate"], errors="coerce"
).dt.to_period("M")

# Merge: each month must match at most one mortgage-rate row
sold_with_rates = sold.merge(
    mortgage_monthly,
    on="year_month",
    how="left",
    validate="many_to_one"
)

listings_with_rates = listings.merge(
    mortgage_monthly,
    on="year_month",
    how="left",
    validate="many_to_one"
)

# Validate row counts and missing rates
assert len(sold_with_rates) == len(sold), "Sold row count changed."
assert len(listings_with_rates) == len(listings), "Listing row count changed."

sold_missing = sold_with_rates["rate_30yr_fixed"].isna().sum()
listings_missing = listings_with_rates["rate_30yr_fixed"].isna().sum()

print("Sold rows:", f"{len(sold_with_rates):,}")
print("Listing rows:", f"{len(listings_with_rates):,}")
print("Sold rows missing mortgage rates:", sold_missing)
print("Listing rows missing mortgage rates:", listings_missing)

print("\nSold preview:")
print(sold_with_rates[
    ["CloseDate", "year_month", "ClosePrice", "rate_30yr_fixed"]
].head())

# Confirm no mortgage rates are missing
assert sold_with_rates["rate_30yr_fixed"].notna().all()
assert listings_with_rates["rate_30yr_fixed"].notna().all()

# Save new CSV files
sold_with_rates.to_csv(
    "/content/combined_sold_with_rates.csv", index=False
)

listings_with_rates.to_csv(
    "/content/combined_listings_with_rates.csv", index=False
)

print("Both enriched datasets saved successfully.")
